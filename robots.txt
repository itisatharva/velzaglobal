# Sitemap line placed at the bottom.

# ------------------------------------------------------------
# ONE GROUP, EVERY CRAWLER
# ------------------------------------------------------------
# robots.txt groups do NOT inherit. A crawler obeys only the single
# most-specific group that names it and ignores every other group,
# including "User-agent: *". Giving each bot its own "Allow: /" group -
# as this file used to - therefore exempted every named bot from the
# Disallow list below: Googlebot was free to crawl /404, /comingsoon
# and /shared/vendor/.
#
# Stacking the user-agent lines in front of ONE rule set is the spec's
# own answer (RFC 9309 s2.2.1: a group is one or more user-agent lines
# followed by rules). Every agent listed here gets the identical policy,
# and the Disallow list stays written exactly once.

User-agent: *

# --- OpenAI / ChatGPT ---
# ChatGPT Search discovery and citation
User-agent: OAI-SearchBot
# OpenAI model crawler
User-agent: GPTBot

# --- Anthropic / Claude ---
# Claude web-search indexing
User-agent: Claude-SearchBot
# User-requested Claude webpage access
User-agent: Claude-User
# Anthropic model-development crawler
User-agent: ClaudeBot

# --- Perplexity ---
# Perplexity search indexing
User-agent: PerplexityBot
# User-requested Perplexity webpage retrieval
User-agent: Perplexity-User

# --- Google / Gemini ---
# Google Search, including Google's AI search experiences
User-agent: Googlebot
# Gemini / Google AI model and grounding controls
User-agent: Google-Extended

# --- Microsoft / Bing / Copilot ---
User-agent: Bingbot

Allow: /

# URLs are extensionless: .html requests 301 to the clean URL,
# so these Disallow paths must be extensionless too.

# Unfinished / placeholder
Disallow: /comingsoon

# Error pages
Disallow: /400
Disallow: /401
Disallow: /403
Disallow: /404
Disallow: /500
Disallow: /502
Disallow: /503
Disallow: /504

# Third-party vendor libraries
Disallow: /shared/vendor/

# Region source directories are served by internal rewrite only
# and return 404 if requested directly.
Disallow: /regions/
